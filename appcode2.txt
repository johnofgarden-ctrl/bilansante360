const questions = [];

const famTopics = [
    "J'ai du mal à exprimer mes émotions librement au sein de mon foyer",
    "Les tensions d'ordre financier créent du stress récurrent dans la famille",
    "La répartition des tâches ménagères ou logistiques engendre des conflits",
    "Je me sens dépassé(e) par la gestion et l'éducation de mes enfants",
    "Le manque de communication avec mon/ma partenaire me pèse au quotidien",
    "Je manque de moments de répit ou de calme dans mon environnement familial",
    "Je dois gérer la dépendance ou les soucis d'un membre proche de ma famille",
    "J'ai le sentiment d'assumer seul(e) la charge mentale de mon foyer",
    "Les désaccords sur les projets de vie créent une distance avec mes proches",
    "Les réunions ou repas de famille sont une source d'anxiété pour moi"
];

for(let i=1; i<=50; i++) {
    const baseTopic = famTopics[(i-1) % famTopics.length];
    questions.push({ id: i, category: 'Familial', text: `${baseTopic} (Indicateur F${i})` });
}

const persoTopics = [
    "J'ai des difficultés régulières d'endormissement ou un sommeil non réparateur",
    "Je ressens des douleurs physiques (dos, nuque, maux de ventre) liées au stress",
    "J'ai tendance à ruminer mes pensées négatives tard le soir",
    "Je consacre très peu de temps à mes loisirs, activités créatives ou sportives",
    "J'ai une baisse significative d'énergie ou d'enthousiasme au quotidien",
    "Je me montre excessivement exigeant(e) et autocritique envers moi-même",
    "J'ai tendance à grignoter, fumer ou consommer davantage sous l'effet du stress",
    "Je me sens souvent submergé(e) par la peur de l'échec ou le doute",
    "J'ai du mal à dire 'non' et à fixer des limites claires aux autres",
    "Je ressens un sentiment d'isolement ou d'incompréhension autour de moi"
];

for(let i=51; i<=100; i++) {
    const baseTopic = persoTopics[(i-51) % persoTopics.length];
    questions.push({ id: i, category: 'Personnel', text: `${baseTopic} (Indicateur P${i-50})` });
}

const proTopics = [
    "Ma charge de travail dépasse régulièrement ma capacité temporelle",
    "Mes objectifs professionnels manquent de clarté ou sont contradictoires",
    "Je ne reçois pas la reconnaissance méritée pour mes efforts et résultats",
    "L'équilibre entre ma vie professionnelle et ma vie personnelle est rompu",
    "Les interruptions et urgences incessantes m'empêchent de me concentrer",
    "J'appréhende la reprise du travail après le week-end ou les vacances",
    "Les outils informatiques et processus internes me causent de la frustration",
    "Je sens un risque d'épuisement professionnel (burn-out) à moyen terme",
    "L'organisation dans mon entreprise manque de stabilité et de vision",
    "Le niveau d'exigence de ma hiérarchie est déconnecté des réalités du terrain"
];

const harassmentTopics = [
    "Je subis des remarques désobligeantes, de l'humiliation ou des moqueries répétées",
    "On me confie volontairement des tâches inutiles ou impossibles à réaliser",
    "Je me sens sciemment mis(e) à l'écart, isolé(e) ou ignoré(e) par des collègues/superieurs",
    "Mes décisions ou mes travaux sont systématiquement critiqués de façon injustifiée",
    "Je fais l'objet de rumeurs, d'attaques personnelles ou de propos diffamatoires",
    "Je subis des pressions morales, du chantage ou des menaces de sanctions injustifiées",
    "On m'enlève progressivement mes responsabilités sans motif professionnel valable",
    "Je ressens de l'agressivité ou des hurlements de la part d'un responsable ou collègue",
    "Je crains pour ma santé mentale directement à cause de l'attitude d'une personne au travail",
    "Je me sens surveillé(e) ou fliqué(e) à un niveau disproportionné"
];

for(let i=101; i<=150; i++) {
    let text = "";
    let isHarassment = false;
    if (i >= 131 && i <= 140) {
        text = harassmentTopics[i - 131];
        isHarassment = true;
    } else {
        const baseTopic = proTopics[(i-101) % proTopics.length];
        text = `${baseTopic} (Indicateur PR${i-100})`;
    }
    questions.push({ id: i, category: 'Professionnel', text: text, isHarassment: isHarassment });
}

let userProfile = {};
let answers = {};
let currentPage = 0;
const PAGE_SIZE = 10;

function startQuiz(e) {
    if (e) e.preventDefault();
    userProfile = {
        name: document.getElementById('user-name').value || 'Utilisateur',
        email: document.getElementById('user-email').value || 'N/A',
        marital: document.getElementById('user-marital').value,
        kids: document.getElementById('user-kids').value,
        role: document.getElementById('user-role').value
    };
    document.getElementById('step-identity').classList.add('hidden');
    document.getElementById('step-quiz').classList.remove('hidden');
    renderPage();
}
